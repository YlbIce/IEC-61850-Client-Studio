cmake_minimum_required(VERSION 3.24)
project(IEC61850Capture LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin")
find_path(JSON_INCLUDE nlohmann/json.hpp REQUIRED)
add_library(iec61850-capture SHARED src/capture.cpp src/process.cpp src/packet_store.cpp)
target_include_directories(iec61850-capture PRIVATE include ${JSON_INCLUDE})
target_compile_definitions(iec61850-capture PRIVATE CAPTURE_BUILD NOMINMAX WIN32_LEAN_AND_MEAN)
add_executable(capture-helper src/helper.cpp)
target_include_directories(capture-helper PRIVATE ${JSON_INCLUDE})
target_compile_definitions(capture-helper PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
foreach(target iec61850-capture capture-helper)
  target_compile_options(${target} PRIVATE /W4 /utf-8 /EHsc)
endforeach()
option(CAPTURE_BUILD_TEST_CLIENT "Build loopback MMS test client against the independently built main backend" OFF)
if(CAPTURE_BUILD_TEST_CLIENT)
  set(STUDIO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
  set(IEC "${STUDIO_ROOT}/third_party/libiec61850")
  add_executable(capture-mms-client tests/mms_client.cpp)
  target_include_directories(capture-mms-client PRIVATE ${IEC}/src/iec61850/inc ${IEC}/src/common/inc ${IEC}/src/mms/inc ${IEC}/src/logging ${IEC}/hal/inc ${IEC}/config)
  target_link_libraries(capture-mms-client PRIVATE ${STUDIO_ROOT}/backend/build-v2/third_party/libiec61850/src/iec61850.lib ${STUDIO_ROOT}/backend/build-v2/third_party/libiec61850/hal/hal.lib ${IEC}/third_party/winpcap/Lib/wpcap.lib ${IEC}/third_party/winpcap/Lib/packet.lib bcrypt)
  set_property(TARGET capture-mms-client PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
endif()
